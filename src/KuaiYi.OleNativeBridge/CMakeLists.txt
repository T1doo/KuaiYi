cmake_minimum_required(VERSION 3.28)
project(KuaiYiOleNativeBridge LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Requires Windows x64 MSVC/SDK; no COM apartment proof on Linux")
endif()
add_library(KuaiYi.OleNativeBridge SHARED DropTargetBridge.cpp)
target_compile_features(KuaiYi.OleNativeBridge PRIVATE cxx_std_20)
target_compile_definitions(KuaiYi.OleNativeBridge PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(KuaiYi.OleNativeBridge PRIVATE /W4 /WX /EHsc)
target_link_libraries(KuaiYi.OleNativeBridge PRIVATE ole32 uuid)
set_property(TARGET KuaiYi.OleNativeBridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
