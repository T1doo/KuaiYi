    internal sealed class Destination:SynchronizationContext
    {
        private readonly ConcurrentQueue<Action> callbacks=new();public volatile bool Hold;public int Pending=>callbacks.Count;
        public override void Post(SendOrPostCallback action,object? state){if(Hold)callbacks.Enqueue(()=>action(state));else action(state);}
        public void Drain(){while(callbacks.TryDequeue(out var action))action();}
    }
    private sealed class SyntheticOcr:P.ILocalOcrWorker
    {
        public Task<P.ImageReply> RunAsync(ReadOnlyMemory<byte> image,P.ImageJob job,CancellationToken cancellation=default)=>Task.FromResult(new P.ImageReply(job,P.ImageFailure.None,new(8,8,job.Language,"PUBLIC SYNTHETIC OCR TEXT",[])));
        public ValueTask<bool> TryFinalizeAsync()=>ValueTask.FromResult(true);
        public ValueTask DisposeAsync()=>ValueTask.CompletedTask;
    }
    public readonly PhaseHttpFixture Server=new();public PhaseHttpFixture.Script Script=null!;
    public readonly ShellWindow Window=new();public RuntimeGate Gate=>Window.MarkdownGate;public OverlayResults Results=>Window.MarkdownResults;
    public readonly ProviderClient Provider=new();public TextTranslationCoordinator Flow=null!;public SettingsSnapshot Settings=null!;
    public readonly Destination Context=new();public readonly ConcurrentQueue<TextTranslationUpdate> Events=new();public Action<TextTranslationUpdate>? OnUpdate;
    public Protocol Protocol;public bool Stream,Current=true;public P.OcrOnlyCoordinator? Local;private P.OwnedSourceImage? image;
    public static async Task<TextUsageRig> Create(Protocol protocol=Protocol.OpenAIChat,bool stream=false,TranslationPreset preset=TranslationPreset.Faithful)
    {
        var r=new TextUsageRig{Protocol=protocol,Stream=stream};await r.Server.InitializeAsync();r.Script=r.Server.Add("runs",stream?"text/event-stream":"application/json");r.Window.InitializeResultFlow();r.Window.WithGeometry();
        var prefix=new Uri(r.Server.BaseUri,$"runs/{Guid.NewGuid():N}/fragmented/");var boundary=DevelopmentTranslationBoundary.FromPrefix(prefix);
        r.Settings=new(new(){ActiveProfileId="text-usage",Profiles=[new(){Id="text-usage",Protocol=protocol==Protocol.OpenAIChat?KuaiYi.Core.Settings.ApiProtocol.OpenAIChat:KuaiYi.Core.Settings.ApiProtocol.AnthropicMessages,BaseUrl=prefix.AbsoluteUri,Model=DevelopmentTranslationBoundary.FixtureModel,Authentication=ApiAuthentication.None,Streaming=stream,Capabilities=new(){UsageInStream=Capability.Supported},TimeoutSeconds=20}],Translation=new(){Preset=preset}},"text-usage-synthetic",SettingsLoadState.Current);
        r.Gate.ObserveSettingsMetadata(r.Settings);r.Gate.ConsentTo(new(prefix));Assert.True(r.Gate.TryEnable(new(prefix),true));
        r.Flow=new(r.Gate,()=>r.Settings,(_,_)=>r.Current,r.Provider,r.Context,r.Results,u=>{r.Window.CoordinatorUpdate(u);r.OnUpdate?.Invoke(u);r.Events.Enqueue(u);},(_,_,_)=>throw new InvalidOperationException("AuthNone fixture must not read credentials."),developmentBoundary:boundary,ocrSourceCurrent:s=>r.Local?.IsApprovedSourceCurrent(s)==true);
        r.Window.UseTranslation(r.Flow,r.Settings);return r;
    }
    public SelectionSourceSnapshot Source(string text="PUBLIC SYNTHETIC SOURCE",SelectionSourceKind kind=SelectionSourceKind.Uia)=>new(new(17,92823,132000,@"C:\Synthetic\fixture-editor.exe"),text,text,[new(1,2,20,22)],[7,8],Settings.Version,text.Length,false,kind,DropReceiver:kind==SelectionSourceKind.OleDrop?Guid.NewGuid():Guid.Empty,DropTransaction:kind==SelectionSourceKind.OleDrop?Guid.NewGuid():Guid.Empty);
    public async Task<SelectionSourceSnapshot> OcrSource()
    {
        image=new(new P.ScreenshotPixels(new(0,0,8,8),Enumerable.Repeat((byte)255,256).ToArray()));var version=Guid.NewGuid();
        Local=new(Gate,()=>Settings,(v,id)=>Current&&v==version&&id==image.SourceId,new Destination(),()=>new SyntheticOcr());
        Assert.True(Local.Start(version,image,"en-US"));await Local.Running.WaitAsync(TimeSpan.FromSeconds(6));var preview=Assert.IsType<P.OcrTextPreview>(Local.Preview);Assert.Equal(0,Script.Requests);return Assert.IsType<SelectionSourceSnapshot>(Local.Approve(preview));
    }
    public RequestIdentity Start(SelectionSourceSnapshot? source=null,bool regenerate=false)
    {
        source??=Source();var offer=regenerate?Flow.PrepareRegeneration():source.SourceKind==SelectionSourceKind.OcrOnly?Flow.PrepareOcr(source,Settings.Data.Translation.Preset):Flow.Prepare(source);Assert.NotNull(offer);
        if(offer.RequiresConfirmation){int before=Script.Requests;Assert.Equal(TextSubmission.ConfirmationRequired,Flow.Submit(offer));Assert.Equal(before,Script.Requests);}
        Assert.Equal(TextSubmission.Started,Flow.Submit(offer,offer.RequiresConfirmation));return Results.Identity!.Value;
    }
    public async Task Requests(int count)=>await Wait(()=>Volatile.Read(ref Script.Requests)>=count);
    public static async Task Wait(Func<bool> condition){using var limit=new CancellationTokenSource(TimeSpan.FromSeconds(6));while(!condition())await Task.Delay(5,limit.Token);}
    public string Counts(long input,long output)=>Protocol==Protocol.OpenAIChat?JsonSerializer.Serialize(new{prompt_tokens=input,completion_tokens=output}):JsonSerializer.Serialize(new{input_tokens=input,output_tokens=output});
    public string OutputOnly(long count)=>"{\""+(Protocol==Protocol.OpenAIChat?"completion_tokens":"output_tokens")+"\":"+count+"}";
    public string UsageFrame(string usage)=>Protocol==Protocol.OpenAIChat?"data: {\"choices\":[],\"usage\":"+usage+"}\n\n":"data: {\"type\":\"message_delta\",\"usage\":"+usage+"}\n\n";
    public string StartFrames=>Protocol==Protocol.OpenAIChat?"data: {\"choices\":[{\"index\":0,\"delta\":{\"role\":\"assistant\"}}]}\n\n":"data: {\"type\":\"message_start\",\"message\":{}}\n\ndata: {\"type\":\"content_block_start\",\"index\":0,\"content_block\":{\"type\":\"text\",\"text\":\"\"}}\n\n";
    public string EndFrames(string text="PUBLIC SYNTHETIC RESULT",bool terminal=true,string finish="stop")
    {
        string quoted=JsonSerializer.Serialize(text);
        return Protocol==Protocol.OpenAIChat?"data: {\"choices\":[{\"index\":0,\"delta\":{\"content\":"+quoted+"}}]}\n\ndata: {\"choices\":[{\"index\":0,\"delta\":{},\"finish_reason\":\""+finish+"\"}]}\n\n"+(terminal?"data: [DONE]\n\n":""):
            "data: {\"type\":\"content_block_delta\",\"index\":0,\"delta\":{\"type\":\"text_delta\",\"text\":"+quoted+"}}\n\ndata: {\"type\":\"content_block_stop\",\"index\":0}\n\ndata: {\"type\":\"message_delta\",\"delta\":{\"stop_reason\":\""+(finish=="stop"?"end_turn":finish)+"\"}}\n\n"+(terminal?"data: {\"type\":\"message_stop\"}\n\n":"");
    }
    public async Task Reply(string? usage,string text="PUBLIC SYNTHETIC RESULT",bool terminal=true,string finish="stop")
    {
        string field=usage==null?"":",\"usage\":"+usage;
        string body=Stream?StartFrames+(usage==null?"":UsageFrame(usage))+EndFrames(text,terminal,finish):Protocol==Protocol.OpenAIChat?
            "{\"choices\":[{\"index\":0,\"message\":{\"content\":"+JsonSerializer.Serialize(text)+"},\"finish_reason\":\""+finish+"\"}]"+field+"}":
            "{\"type\":\"message\",\"content\":[{\"type\":\"text\",\"text\":"+JsonSerializer.Serialize(text)+"}],\"stop_reason\":\""+(finish=="stop"?"end_turn":finish)+"\""+field+"}";
        await Script.Write(body,true);await Flow.Running.WaitAsync(TimeSpan.FromSeconds(6));
    }
    public CachedOverlayVersion Version(RequestIdentity identity)=>Results.Versions.Single(v=>v.Identity==identity);
    public async ValueTask DisposeAsync(){Gate.Disable();Context.Hold=false;Context.Drain();await Flow.DisposeAsync();if(Local!=null)await Local.DisposeAsync();image?.Dispose();Provider.Dispose();await Server.DisposeAsync();}
}

    public static IEnumerable<object[]> QueuedChanges()

    [Theory][MemberData(nameof(QueuedChanges))]
    public async Task QueuedActualUsageCannotOutliveIdentitySourceOrSettings(Protocol protocol,string action)
    {
        await using var r=await TextUsageRig.Create(protocol,true);var oldId=r.Start();var old=r.Flow.Running;await r.Requests(1);r.Context.Hold=true;await r.Script.Write(r.StartFrames+r.UsageFrame(r.Counts(99,88)));await TextUsageRig.Wait(()=>r.Context.Pending>0);
        RequestIdentity? newer=null;
        switch(action)
        {
            case "clear":Assert.True(await r.Window.ClearContentForTest());break;
            case "off":r.Gate.Disable();break;case "lock":r.Gate.SetSessionLocked(true);break;case "hide":r.Gate.Disable(hide:true);break;case "stop":r.Window.StopForTest();break;
            case "mode":r.Gate.SetMode(TriggerMode.Drop);break;

            case "source":r.Current=false;break;
            case "new-source":newer=r.Start(r.Source("PUBLIC NEW C"));break;
        }
        r.Context.Hold=false;r.Context.Drain();Assert.DoesNotContain(r.Events,e=>e.Presentation.Kind==OverlayResultEventKind.Usage);r.Gate.StopRequest(oldId);await old.WaitAsync(TimeSpan.FromSeconds(6));
        if(newer is {} next){await r.Requests(2);await r.Reply(r.Counts(3,2));Assert.Equal(new TranslationUsage(3,2),r.Version(next).Usage);Assert.Equal(next,r.Results.Identity);Assert.Single(r.Events,e=>e.Presentation.Kind==OverlayResultEventKind.Usage);}else if(action=="clear"){Assert.Empty(r.Results.Versions);Assert.Empty(r.Window.ResultLabel);}else Assert.Equal(TranslationUsage.Unknown,r.Version(oldId).Usage);
