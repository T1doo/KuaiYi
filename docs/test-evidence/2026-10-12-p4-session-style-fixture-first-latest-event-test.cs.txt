    private async Task<CapturedRequest[]> Requests(Rig rig){using var http=new HttpClient();return(await http.GetFromJsonAsync<CapturedRequest[]>(new Uri(fixture.BaseUri,$"requests/{rig.Run}")))!;}
    private static string SystemPrompt(CapturedRequest request,Protocol protocol){using var json=JsonDocument.Parse(request.Body);return protocol==Protocol.OpenAIChat?json.RootElement.GetProperty("messages")[0].GetProperty("content").GetString()!:json.RootElement.GetProperty("system").GetString()!;}
    [Theory] [InlineData(Protocol.OpenAIChat)] [InlineData(Protocol.AnthropicMessages)]
    public async Task ActualRegenerationAddsFactPreservationConstraintOnce(Protocol protocol)
    {
        await using var rig=new Rig(fixture.BaseUri,protocol);await rig.Start();await rig.Regenerate();await rig.Regenerate();var requests=await Requests(rig);Assert.Equal(3,requests.Length);Assert.DoesNotContain(Constraint,SystemPrompt(requests[0],protocol));
        foreach(var request in requests.Skip(1)){var prompt=SystemPrompt(request,protocol);Assert.Contains(Constraint,prompt);Assert.Equal(1,prompt.Split(Constraint).Length-1);using var body=JsonDocument.Parse(request.Body);Assert.Equal("PUBLIC_SESSION_MODEL",body.RootElement.GetProperty("model").GetString());}
    }

    private async Task Barrier(Rig rig)
    {using var http=new HttpClient();using var deadline=new CancellationTokenSource(TimeSpan.FromSeconds(6));while(true){var value=await http.GetFromJsonAsync<JsonElement>(new Uri(fixture.BaseUri,$"barriers/{rig.Run}"),deadline.Token);if(value.GetProperty("ready").GetBoolean())return;await Task.Delay(10,deadline.Token);}}
    private async Task Release(Rig rig){using var http=new HttpClient();using var reply=await http.PostAsync(new Uri(fixture.BaseUri,$"barriers/{rig.Run}/release"),null);reply.EnsureSuccessStatusCode();}

        await using var rig=new Rig(fixture.BaseUri,protocol);await rig.Start();var old=rig.Flow.SelectedSnapshot!;var oldId=rig.Results.SelectedVersionId!.Value;var global=rig.Settings;

        var offer=rig.Flow.PrepareRegeneration()!;Assert.Equal(2,offer.Snapshot.VersionNumber);Assert.Equal(preset,offer.Snapshot.Preset);Assert.Equal("regenerate@1",offer.Snapshot.RegenerationTemplateVersion);Assert.EndsWith("+regenerate@1",offer.Snapshot.FullTemplateVersion);Assert.Same(old.Profile,offer.Snapshot.Profile);Assert.Equal(old.Transport,offer.Snapshot.Transport);

    private sealed class QueuedDestination:SynchronizationContext
    {public bool Hold=true;public TaskCompletionSource Entered=new(TaskCreationOptions.RunContinuationsAsynchronously);private readonly System.Collections.Concurrent.ConcurrentQueue<Action> queue=new();public override void Post(SendOrPostCallback callback,object? state){if(!Hold){callback(state);return;}queue.Enqueue(()=>callback(state));Entered.TrySetResult();}public void Drain(){Hold=false;while(queue.TryDequeue(out var callback))callback();}}
    [Theory] [InlineData("off")] [InlineData("replacement")]
    public async Task DelayedDestinationCannotRelabelOrOverwriteAnotherVersion(string action)
    {
        var destination=new QueuedDestination();await using var rig=new Rig(fixture.BaseUri,destination:destination);rig.Flow.Submit(rig.Flow.Prepare(rig.Source())!);var old=rig.Flow.Running;await destination.Entered.Task.WaitAsync(TimeSpan.FromSeconds(5));rig.Flow.SetSessionStyle(TranslationPreset.Professional);
        if(action=="off")rig.Gate.Disable();else rig.Gate.StopGeneration();await old.WaitAsync(TimeSpan.FromSeconds(5));destination.Drain();Assert.Empty(rig.Results.Text);Assert.DoesNotContain(rig.Updates,u=>u.Presentation.Kind!=OverlayResultEventKind.Begin);
        if(action=="replacement"){await rig.Start("PUBLIC another source");Assert.Equal(TranslationPreset.Faithful,rig.Flow.SelectedSnapshot!.Preset);Assert.Null(rig.Flow.SessionStyle);Assert.Equal("PUBLIC another source",rig.Flow.SelectedSnapshot.Source.Text);}
