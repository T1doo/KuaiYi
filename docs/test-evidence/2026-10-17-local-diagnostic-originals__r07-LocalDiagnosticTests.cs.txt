    private sealed class DirectoryOwner:IDisposable
    {

        public void Dispose()=>Directory.Delete(Path,true);

    private sealed class HostileException(string payload):Exception(payload)
    {public override string Message=>throw new InvalidOperationException("Message must not be read.");public override string ToString()=>throw new InvalidOperationException("Exception must not be formatted.");}
    [Fact]public void BoundedSessionRetainsNewestMetadataAndClearsOnDispose()
    {
        var clock=new Clock();using var log=new DiagnosticLog(new(1,2,3),clock);
        for(int n=0;n<1000;n++){clock.Now=clock.Now.AddMilliseconds(1);Assert.True(log.Record(DiagnosticEvent.CaptureFinished,byteCount:n));}
        var snapshot=log.Snapshot();Assert.Equal(256,snapshot.Length);Assert.Equal(744,snapshot[0].ByteCount);Assert.Equal(999,snapshot[^1].ByteCount);Assert.Equal(clock.Now,snapshot[^1].Utc);
        log.Dispose();Assert.Empty(log.Snapshot());Assert.False(log.Record(DiagnosticEvent.Started));
    }
    [Fact]public void ConcurrentEventsCannotExceedSessionBound()
    {using var log=new DiagnosticLog(new(1,0));Parallel.For(0,5000,n=>Assert.True(log.Record(DiagnosticEvent.ApplicationFailure,DiagnosticError.Io,byteCount:n)));Assert.Equal(256,log.Snapshot().Length);log.Clear();Assert.Empty(log.Snapshot());}

    [Fact]public async Task RealExportOfFailureReportCannotContainSyntheticSecretBodyPathOrQuery()
    {
        using var directory=new DirectoryOwner();using var log=new DiagnosticLog(new(1,2,3));
        string[] sentinels=["PUBLIC_FAKE_KEY_sk-never-real","PUBLIC_SOURCE_BODY","PUBLIC_TRANSLATION_BODY","PUBLIC_OCR_BODY",@"C:\Users\PUBLIC\private.png","https://public.invalid/api?token=PUBLIC_QUERY"];
        string payload=string.Join("\n",sentinels);
        foreach(var error in new Exception[]{new IOException(payload),new JsonException(payload),new UnauthorizedAccessException(payload),new ArgumentException(payload),new AggregateException(new Exception(payload)),new HostileException(payload)})Assert.True(log.RecordFailure(error));
        var id=Guid.NewGuid();Assert.True(log.Record(DiagnosticEvent.TranslationFinished,DiagnosticError.Provider,id,123,456,1));
        using var session=new DiagnosticReportSession();var report=session.Prepare(log,EnvironmentData());
        Assert.Contains("AppVersion=1.2.3",report.Text);Assert.Contains("OS=Linux Build=6.8.1",report.Text);Assert.Contains("Monitor[0]=-1920,0,1920,1080 DPI=144,144",report.Text);Assert.Contains("Component=Core Version=1.2.3",report.Text);Assert.Contains("Capability=RealModel Result=NotRun",report.Text);Assert.Contains(id.ToString("D"),report.Text);
        string path=System.IO.Path.Combine(directory.Path,"report.txt");Assert.Equal(DiagnosticExportStatus.Published,await session.ExportAsync(report,(r,ct)=>DiagnosticReportFiles.PrepareNewAsync(path,r,ct)));
        byte[] exported=await File.ReadAllBytesAsync(path);Assert.Equal(Encoding.UTF8.GetBytes(report.Text),exported);
        foreach(string sentinel in sentinels)Assert.DoesNotContain(sentinel,Encoding.UTF8.GetString(exported));
        Assert.Single(Directory.GetFiles(directory.Path));Assert.Equal(7,log.Snapshot().Length);
    }

    [Theory][InlineData("cancel")][InlineData("invalidate")][InlineData("dispose")]
    public async Task LatePreparedFileAfterCancellationOrCloseIsCleanedWithoutPublication(string reason)
    {
        using var directory=new DirectoryOwner();using var log=new DiagnosticLog(new(1,0));using var session=new DiagnosticReportSession();using var cancellation=new CancellationTokenSource();var report=session.Prepare(log,EnvironmentData());string path=System.IO.Path.Combine(directory.Path,"report.txt");
        var ready=new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);var release=new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);
        var exporting=session.ExportAsync(report,async(r,ct)=>{var file=await DiagnosticReportFiles.PrepareNewAsync(path,r,ct);ready.SetResult();await release.Task;return file;},cancellation.Token);
        await ready.Task.WaitAsync(TimeSpan.FromSeconds(5));Assert.True(session.Busy);Assert.Equal(DiagnosticExportStatus.Busy,await session.ExportAsync(report,(_,_)=>throw new InvalidOperationException("Must not prepare a second writer.")));
        if(reason=="cancel")cancellation.Cancel();else if(reason=="dispose")session.Dispose();else session.Invalidate();release.SetResult();Assert.Equal(DiagnosticExportStatus.Cancelled,await exporting);Assert.False(session.Busy);Assert.Empty(Directory.GetFiles(directory.Path));
    }
    [Fact]public async Task OldPreviewAndPrecancelCannotCreateFilesAndSuccessfulPreviewIsFrozen()
    {
        using var directory=new DirectoryOwner();using var log=new DiagnosticLog(new(1,0));using var session=new DiagnosticReportSession();var old=session.Prepare(log,EnvironmentData());log.Record(DiagnosticEvent.ApplicationFailure,DiagnosticError.Timeout);var current=session.Prepare(log,EnvironmentData());Assert.NotEqual(old.Text,current.Text);
        Assert.Equal(DiagnosticExportStatus.Cancelled,await session.ExportAsync(old,(_,_)=>throw new InvalidOperationException("Old preview cannot export.")));
        using var cancel=new CancellationTokenSource();cancel.Cancel();Assert.Equal(DiagnosticExportStatus.Cancelled,await session.ExportAsync(current,(_,_)=>throw new InvalidOperationException("Precancel cannot export."),cancel.Token));
        log.Clear();string path=System.IO.Path.Combine(directory.Path,"report.txt");Assert.Equal(DiagnosticExportStatus.Published,await session.ExportAsync(current,(r,ct)=>DiagnosticReportFiles.PrepareNewAsync(path,r,ct)));Assert.Equal(current.Text,await File.ReadAllTextAsync(path));Assert.Contains("Error=Timeout",current.Text);
    }

    [Theory][InlineData(false)][InlineData(true)]
    public async Task CancellationCallbackCannotPreventPreviewRevocationOrBlockTheCaller(bool throwing)
    {
        using var directory=new DirectoryOwner();using var log=new DiagnosticLog(new(1,0));using var session=new DiagnosticReportSession();var report=session.Prepare(log,EnvironmentData());
        var ready=new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);var release=new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);
        var entered=new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);var finishCallback=new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);
        string target=System.IO.Path.Combine(directory.Path,"report.txt");
        var exporting=session.ExportAsync(report,async(r,ct)=>
        {
            var file=await DiagnosticReportFiles.PrepareNewAsync(target,r,ct);
            ct.Register(()=>{entered.TrySetResult();if(throwing)throw new IOException("PUBLIC_FAKE_KEY_AND_BODY");finishCallback.Task.GetAwaiter().GetResult();});
            ready.SetResult();await release.Task;return file;
        });
        await ready.Task.WaitAsync(TimeSpan.FromSeconds(5));var revoke=Task.Run(()=>Record.Exception(session.Invalidate));
        Exception? cancellationError=null;bool returned=false;
        try{await entered.Task.WaitAsync(TimeSpan.FromSeconds(5));cancellationError=await revoke.WaitAsync(TimeSpan.FromSeconds(1));returned=true;}
        catch(TimeoutException){ }
        finally{finishCallback.TrySetResult();release.TrySetResult();}
        await revoke;Assert.Equal(DiagnosticExportStatus.Cancelled,await exporting);Assert.Null(session.Preview);Assert.Empty(Directory.GetFiles(directory.Path));Assert.True(returned,"Preview invalidation must not wait for cancellation callbacks.");Assert.Null(cancellationError);
    }
