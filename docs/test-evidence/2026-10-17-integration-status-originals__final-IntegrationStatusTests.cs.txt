    private sealed class Installation:IDisposable

        public readonly string DirectoryPath=Path.Combine(Path.GetTempPath(),"KuaiYi.IntegrationFixture."+Guid.NewGuid().ToString("N"));

        public readonly Dictionary<string,string> Hashes=new();
        public Installation()
        {
            Directory.CreateDirectory(DirectoryPath);
            foreach(string name in InstalledRoleFiles.Required){byte[] value=System.Text.Encoding.UTF8.GetBytes("public synthetic role "+name);File.WriteAllBytes(Path.Combine(DirectoryPath,name),value);Hashes.Add(name,Convert.ToHexString(SHA256.HashData(value)));}

            SaveReceipt();
        }
        public void SaveReceipt(int schema=1)=>File.WriteAllText(Receipt,JsonSerializer.Serialize(new{schemaVersion=schema,files=Hashes}));
        public void Dispose()=>Directory.Delete(DirectoryPath,true);
    }

    [Theory][InlineData("missing",IntegrationState.NotInstalled,IntegrationReason.RegistrationMissing)][InlineData("different",IntegrationState.Incompatible,IntegrationReason.RegistrationMismatch)][InlineData("conflict",IntegrationState.Unavailable,IntegrationReason.EvidenceConflict)][InlineData("permission",IntegrationState.Unavailable,IntegrationReason.PermissionDenied)][InlineData("io",IntegrationState.Unavailable,IntegrationReason.ReadFailed)][InlineData("invalid",IntegrationState.Incompatible,IntegrationReason.RegistrationMismatch)][InlineData("duplicate-source",IntegrationState.Unavailable,IntegrationReason.EvidenceConflict)][InlineData("machine-only",IntegrationState.NotInstalled,IntegrationReason.RegistrationMissing)][InlineData("policy-invalid",IntegrationState.NotChecked,IntegrationReason.PolicyUnknown)][InlineData("policy-permission",IntegrationState.Unavailable,IntegrationReason.PermissionDenied)]
    public void ReadFailuresMissingAndConflictsRemainDistinct(string fault,IntegrationState state,IntegrationReason reason)
    {
        string directory=Path.GetFullPath("public-installation");var rows=Registry(directory,fault!="missing"&&fault!="machine-only");
        rows=fault switch
        {
            "different"=>rows.SetItem(0,rows[0] with{Registration=new(IntegrationValueState.Present,"\\\\unvisited-server\\share\\manifest.json")}),
            "conflict"=>rows.SetItem(1,rows[1] with{Registration=new(IntegrationValueState.Present,Path.Combine(directory,"other.json"))}),
            "permission"=>rows.SetItem(1,rows[1] with{Registration=new(IntegrationValueState.PermissionDenied)}),
            "io"=>rows.SetItem(3,rows[3] with{Registration=new(IntegrationValueState.Failed)}),
            "invalid"=>rows.SetItem(0,rows[0] with{Registration=new(IntegrationValueState.Invalid)}),
            "duplicate-source"=>rows.SetItem(1,rows[0]),

    [Theory][InlineData("block",IntegrationState.PolicyDenied,IntegrationReason.LocalPolicyBlocked)][InlineData("block-exemption",IntegrationState.Configured,IntegrationReason.LocalConfigurationOnly)][InlineData("user-disabled",IntegrationState.PolicyDenied,IntegrationReason.UserHostsDisabled)][InlineData("user-disabled-machine",IntegrationState.NotChecked,IntegrationReason.PolicyUnknown)][InlineData("conflict",IntegrationState.NotChecked,IntegrationReason.EvidenceConflict)][InlineData("cross-source-exemption",IntegrationState.NotChecked,IntegrationReason.EvidenceConflict)][InlineData("cross-source-complement",IntegrationState.NotChecked,IntegrationReason.EvidenceConflict)][InlineData("permission",IntegrationState.Unavailable,IntegrationReason.PermissionDenied)]
    public void PolicyDeniedRequiresExplicitUnambiguousPolicy(string fault,IntegrationState state,IntegrationReason reason)
    {
        string directory=Path.GetFullPath("public-installation");var rows=Registry(directory);
        rows=rows.SetItem(0,rows[0] with{Policy=fault switch{"block"=>new(IntegrationValueState.Present,HostBlocked:true),"block-exemption"=>new(IntegrationValueState.Present,HostBlocked:true,HostAllowed:true),"permission"=>new(IntegrationValueState.PermissionDenied),_=>new(IntegrationValueState.Present,UserHostsAllowed:false)}});
        if(fault=="conflict")rows=rows.SetItem(2,rows[2] with{Policy=new(IntegrationValueState.Present,UserHostsAllowed:true)});
        if(fault is "cross-source-exemption" or "cross-source-complement")
        {rows=rows.SetItem(0,rows[0] with{Policy=new(IntegrationValueState.Present,HostBlocked:true)});rows=rows.SetItem(2,rows[2] with{Policy=fault=="cross-source-exemption"?new(IntegrationValueState.Present,HostAllowed:true):new(IntegrationValueState.Present,UserHostsAllowed:true)});}

        Assert.Equal((state,reason),InstalledIntegrationStatusReader.ObserveRegistration(rows,directory));
    }

    [Theory][InlineData("manifest-missing",IntegrationState.NotInstalled,IntegrationReason.ManifestMissing)][InlineData("receipt-missing",IntegrationState.NotInstalled,IntegrationReason.ReceiptMissing)][InlineData("manifest-unknown",IntegrationState.Incompatible,IntegrationReason.InvalidManifest)][InlineData("manifest-duplicate",IntegrationState.Incompatible,IntegrationReason.InvalidManifest)][InlineData("manifest-origin",IntegrationState.Incompatible,IntegrationReason.InvalidManifest)][InlineData("manifest-type",IntegrationState.Incompatible,IntegrationReason.InvalidManifest)][InlineData("manifest-path",IntegrationState.Incompatible,IntegrationReason.InvalidManifest)][InlineData("manifest-large",IntegrationState.Incompatible,IntegrationReason.InvalidManifest)][InlineData("receipt-schema",IntegrationState.Incompatible,IntegrationReason.InvalidReceipt)][InlineData("receipt-type",IntegrationState.Incompatible,IntegrationReason.InvalidReceipt)][InlineData("receipt-duplicate",IntegrationState.Incompatible,IntegrationReason.InvalidReceipt)][InlineData("receipt-unknown",IntegrationState.Incompatible,IntegrationReason.InvalidReceipt)][InlineData("receipt-missing-role",IntegrationState.Incompatible,IntegrationReason.InvalidReceipt)][InlineData("receipt-bad-hash",IntegrationState.Incompatible,IntegrationReason.InvalidReceipt)][InlineData("role-missing",IntegrationState.Incompatible,IntegrationReason.FileIntegrity)][InlineData("role-changed",IntegrationState.Incompatible,IntegrationReason.FileIntegrity)][InlineData("role-large",IntegrationState.Incompatible,IntegrationReason.FileIntegrity)]
    public async Task FixedOwnFilesAreStrictBoundedAndDoNotRunCode(string fault,IntegrationState state,IntegrationReason reason)
    {
        using var installation=new Installation();string manifest=File.ReadAllText(installation.Manifest),receipt=File.ReadAllText(installation.Receipt);
        switch(fault)
        {
            case "manifest-missing":File.Delete(installation.Manifest);break;
            case "receipt-missing":File.Delete(installation.Receipt);break;
            case "manifest-unknown":File.WriteAllText(installation.Manifest,manifest[..^1]+",\"unknown\":1}");break;
            case "manifest-duplicate":File.WriteAllText(installation.Manifest,manifest[..^1]+",\"na\\u006de\":\"com.kuaiyi.native_host\"}");break;
            case "manifest-origin":File.WriteAllText(installation.Manifest,manifest.Replace(new string('a',32),"*"));break;
            case "manifest-type":File.WriteAllText(installation.Manifest,manifest.Replace("\"stdio\"","1"));break;
            case "manifest-path":File.WriteAllText(installation.Manifest,manifest.Replace("KuaiYi.NativeHost.exe","Other.exe"));break;
            case "manifest-large":File.WriteAllText(installation.Manifest,new string(' ',65537));break;
            case "receipt-schema":installation.SaveReceipt(2);break;
            case "receipt-type":File.WriteAllText(installation.Receipt,receipt.Replace("\"schemaVersion\":1","\"schemaVersion\":\"1\""));break;
            case "receipt-duplicate":File.WriteAllText(installation.Receipt,receipt[..^1]+",\"schemaVersion\":1}");break;
            case "receipt-unknown":installation.Hashes.Add("../Other.exe",new string('0',64));installation.SaveReceipt();break;
            case "receipt-missing-role":installation.Hashes.Remove("KuaiYi.App.dll");installation.SaveReceipt();break;
            case "receipt-bad-hash":installation.Hashes["KuaiYi.App.dll"]="invalid";installation.SaveReceipt();break;
            case "role-missing":File.Delete(Path.Combine(installation.DirectoryPath,"KuaiYi.App.dll"));break;
            case "role-changed":File.WriteAllText(Path.Combine(installation.DirectoryPath,"KuaiYi.App.dll"),"synthetic modification");break;
            case "role-large":using(var file=File.OpenWrite(Path.Combine(installation.DirectoryPath,"KuaiYi.App.dll")))file.SetLength(IntegrationInstalledFiles.RoleFileLimit+1);break;
        }
        Assert.Equal((state,reason),await IntegrationInstalledFiles.ReadAsync(installation.DirectoryPath,default));
    }

    [Theory][InlineData(false)][InlineData(true)]
    public async Task InvalidateOrCloseRejectsLateResultAndRetainsSingleReadUntilActualSettlement(bool close)
    {
        using var session=new IntegrationStatusSession();var entered=new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);var release=new TaskCompletionSource<IntegrationSnapshot>(TaskCreationOptions.RunContinuationsAsynchronously);int reads=0;
        var run=session.RefreshAsync(_=>{Interlocked.Increment(ref reads);entered.SetResult();return release.Task;});await entered.Task.WaitAsync(TimeSpan.FromSeconds(3));
        if(close)session.Dispose();else session.Invalidate();
        Assert.True(session.Busy);Assert.Null(session.Snapshot);Assert.Equal(close?IntegrationRefreshStatus.Cancelled:IntegrationRefreshStatus.Busy,await session.RefreshAsync(_=>{Interlocked.Increment(ref reads);return Task.FromResult(IntegrationSnapshot.Unsupported);}));
        Assert.False(run.IsCompleted);Assert.Equal(1,reads);release.SetResult(IntegrationSnapshot.Unsupported);Assert.Equal(IntegrationRefreshStatus.Cancelled,await run.WaitAsync(TimeSpan.FromSeconds(3)));Assert.False(session.Busy);Assert.Null(session.Snapshot);
        if(!close){Assert.Equal(IntegrationRefreshStatus.Published,await session.RefreshAsync(_=>Task.FromResult(IntegrationSnapshot.Initial)));Assert.Same(IntegrationSnapshot.Initial,session.Snapshot);}
    }
    [Fact]public async Task CallerCancellationCannotPublishEvenBeforeBackgroundCancellationCallbackRuns()
    {
        using var session=new IntegrationStatusSession();using var caller=new CancellationTokenSource();var entered=new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);var release=new TaskCompletionSource<IntegrationSnapshot>(TaskCreationOptions.RunContinuationsAsynchronously);
        var run=session.RefreshAsync(_=>{entered.SetResult();return release.Task;},caller.Token);await entered.Task.WaitAsync(TimeSpan.FromSeconds(3));caller.Cancel();release.SetResult(IntegrationSnapshot.Unsupported);

    }
    [Fact]public async Task BlockingCancellationCallbackKeepsOwnershipAndDoesNotBlockInvalidation()
    {
        using var session=new IntegrationStatusSession();var entered=new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);var callbackEntered=new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);var callbackRelease=new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);var readRelease=new TaskCompletionSource<IntegrationSnapshot>(TaskCreationOptions.RunContinuationsAsynchronously);
        var run=session.RefreshAsync(async ct=>{using var registration=ct.Register(()=>{callbackEntered.SetResult();callbackRelease.Task.Wait(TimeSpan.FromSeconds(4));});entered.SetResult();return await readRelease.Task;});
        try
        {
            await entered.Task.WaitAsync(TimeSpan.FromSeconds(3));session.Invalidate();await callbackEntered.Task.WaitAsync(TimeSpan.FromSeconds(3));readRelease.SetResult(IntegrationSnapshot.Unsupported);
            Assert.True(session.Busy);Assert.False(run.IsCompleted);Assert.Equal(IntegrationRefreshStatus.Busy,await session.RefreshAsync(_=>Task.FromResult(IntegrationSnapshot.Initial)));
        }
        finally{callbackRelease.TrySetResult();readRelease.TrySetResult(IntegrationSnapshot.Unsupported);}
        Assert.Equal(IntegrationRefreshStatus.Cancelled,await run.WaitAsync(TimeSpan.FromSeconds(3)));Assert.Null(session.Snapshot);
    }
