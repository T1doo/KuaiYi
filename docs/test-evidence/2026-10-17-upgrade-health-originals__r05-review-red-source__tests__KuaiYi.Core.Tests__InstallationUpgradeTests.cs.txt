    private static string Host=>Environment.GetEnvironmentVariable("DOTNET_HOST_PATH")??"dotnet";
    private sealed class Rig:IDisposable
    {
        internal readonly string Root=Path.Combine(Path.GetTempPath(),"kuaiyi-upgrade-tests-"+Guid.NewGuid().ToString("N"));
        internal string App=>Path.Combine(Root,"application");internal string Data=>Path.Combine(Root,"user-data");
        internal InstallationStore Store=>new(App,Data,()=>true);
        internal Rig(){Directory.CreateDirectory(Data);File.WriteAllText(Path.Combine(Data,"config.json"),"PUBLIC_PRESERVED_CONFIG");File.WriteAllBytes(Path.Combine(Data,"encrypted-secret.bin"),[19,82,7,0,244,11,38]);}
        internal string Package(string version,string mode="healthy")
        {
            string package=Path.Combine(Root,"package-"+version);Directory.CreateDirectory(package);
            foreach(string name in new[]{"KuaiYi.SettingsStoreFixture.dll","KuaiYi.SettingsStoreFixture.runtimeconfig.json","KuaiYi.SettingsStoreFixture.deps.json","KuaiYi.Core.dll"})File.Copy(Path.Combine(Path.GetDirectoryName(Driver)!,name),Path.Combine(package,name));
            File.WriteAllText(Path.Combine(package,"KuaiYi.App.exe"),"PUBLIC_DOTNET_TEST_PORT_ONLY");File.WriteAllText(Path.Combine(package,"KuaiYi.CaptureBroker.exe"),"PUBLIC_DOTNET_TEST_PORT_ONLY");
            File.WriteAllBytes(Path.Combine(package,"public-install-behavior.json"),JsonSerializer.SerializeToUtf8Bytes(new{version,mode}));
            var files=Directory.GetFiles(package).Order().Select(p=>new{path=Path.GetFileName(p),bytes=new FileInfo(p).Length,sha256=Hash(p)}).ToArray();
            File.WriteAllBytes(Path.Combine(package,"package.json"),JsonSerializer.SerializeToUtf8Bytes(new{schemaVersion=1,product="KuaiYi",version,files}));return package;
        }
        internal string Pin(string version)=>Hash(Path.Combine(Root,"package-"+version,"package.json"));
        internal byte[] Pointer=>File.ReadAllBytes(Path.Combine(App,"current.json"));
        internal static string Hash(string path)=>Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(path)));
        internal Dictionary<string,string> DataBytes()=>Directory.GetFiles(Data,"*",SearchOption.AllDirectories).ToDictionary(p=>Path.GetRelativePath(Data,p),Hash);
        internal InstallationUpgradeCoordinator Coordinator(Port port)=>new(App,Data,port,TimeSpan.FromSeconds(2),port);
        public void Dispose(){if(Directory.Exists(Root))Directory.Delete(Root,true);}
    }
    private sealed class Child
    {
        internal readonly Process Process;internal readonly string Version,Directory,DriverHash;
        internal readonly TaskCompletionSource<string> First=new(TaskCreationOptions.RunContinuationsAsynchronously);
        internal readonly Task Output,Error;internal readonly List<string> Lines=[];
        internal bool Settled;internal int Exit;
        internal Child(Process process,string directory,string version,Task? drain=null)
        {
            Process=process;Directory=directory;Version=version;DriverHash=Rig.Hash(Path.Combine(directory,"KuaiYi.SettingsStoreFixture.dll"));
            Output=ReadOutput();Error=ReadError(drain);
        }
        private async Task ReadOutput()
        {
            try
            {
                while(await Process.StandardOutput.ReadLineAsync() is {} line)
                {if(line.Length>1024||Lines.Count>=16)throw new IOException("Public fixture output overflow.");Lines.Add(line);First.TrySetResult(line);}
                First.TrySetException(new IOException("Public fixture exited before ready."));
            }
            catch(Exception error){First.TrySetException(error);throw;}
        }
        private async Task ReadError(Task? drain){string text=await Process.StandardError.ReadToEndAsync();if(drain!=null)await drain;if(text.Length!=0)throw new IOException("Public fixture emitted stderr: "+text);}
    }
    private sealed class Port(Rig rig,ITestOutputHelper output):IInstallationApplicationProtocol,IOwnedProcessStarter
    {
        internal readonly List<Child> Children=[];
        internal string? HoldVersion,FailStartVersion,HoldDrainVersion;
        internal bool IgnoreHealthCancellation;
        internal readonly TaskCompletionSource Held=new(TaskCreationOptions.RunContinuationsAsynchronously),Release=new(TaskCreationOptions.RunContinuationsAsynchronously);
        internal readonly TaskCompletionSource Settling=new(TaskCreationOptions.RunContinuationsAsynchronously),DrainRelease=new(TaskCreationOptions.RunContinuationsAsynchronously);
        public Process Start(ProcessStartInfo info)
        {
            string directory=Path.GetDirectoryName(info.FileName)!;Assert.Equal(directory,info.WorkingDirectory);
            Assert.StartsWith(Path.Combine(rig.App,"versions")+Path.DirectorySeparatorChar,directory,StringComparison.Ordinal);
            Assert.Contains(Path.GetFileName(info.FileName),new[]{"KuaiYi.App.exe","KuaiYi.CaptureBroker.exe"});
            using var config=JsonDocument.Parse(File.ReadAllBytes(Path.Combine(directory,"public-install-behavior.json")));string version=config.RootElement.GetProperty("version").GetString()!;
            if(FailStartVersion==version)info.FileName=Path.Combine(directory,"PUBLIC_MISSING_STARTER");
            else{info.FileName=Host;info.ArgumentList.Insert(0,Path.Combine(directory,"KuaiYi.SettingsStoreFixture.dll"));}
            var child=Process.Start(info)??throw new IOException("Public fixture start failed.");Children.Add(new(child,directory,version,HoldDrainVersion==version?DrainRelease.Task:null));return child;
        }
        public async Task ConfirmHealthyAsync(Process owned,InstalledVersion version,Guid launch,CancellationToken cancellation)
        {
            var child=Children.Single(c=>ReferenceEquals(c.Process,owned));string line=await child.First.Task.WaitAsync(cancellation);string[] fields=line.Split('|');
            if(fields.Length!=6||fields[0]!="PUBLIC_INSTALL_READY"||fields[1]!=launch.ToString("N")||fields[2]!=version.Version||fields[3]!=child.DriverHash||fields[4]!=version.DirectoryPath||fields[5]!="Healthy")throw new InvalidDataException("Public selected-child health mismatch.");
            if(HoldVersion==version.Version){Held.TrySetResult();if(IgnoreHealthCancellation)await Release.Task;else await Release.Task.WaitAsync(cancellation);}
        }
        public async Task RequestExitAsync(Process owned,CancellationToken cancellation)
        {await owned.StandardInput.WriteLineAsync("STOP".AsMemory(),cancellation);await owned.StandardInput.FlushAsync(cancellation);}
        public async Task SettleAsync(Process owned,CancellationToken cancellation)
        {
            var child=Children.Single(c=>ReferenceEquals(c.Process,owned));if(child.Version==HoldDrainVersion)Settling.TrySetResult();await Task.WhenAll(child.Output,child.Error).WaitAsync(cancellation);Assert.True(owned.HasExited);
            if(!child.Settled){child.Settled=true;child.Exit=owned.ExitCode;output.WriteLine($"PUBLIC_UPGRADE_CHILD version={child.Version} directory={child.Directory} driverSha256={child.DriverHash} exit={child.Exit} stdoutLines={child.Lines.Count} stderrBytes=0 IO=Settled");}
        }
        internal void AllSettled()=>Assert.All(Children,c=>{Assert.True(c.Settled);Assert.True(c.Output.IsCompletedSuccessfully);Assert.True(c.Error.IsCompletedSuccessfully);});
    }
    private async Task AdmissionChild(Rig rig,bool admitted)
    {
        var info=new ProcessStartInfo(Host){UseShellExecute=false,RedirectStandardOutput=true,RedirectStandardError=true};foreach(string arg in new[]{Driver,"--installation-admission",rig.Root})info.ArgumentList.Add(arg);
        using var child=Process.Start(info)!;Task<string> stdout=child.StandardOutput.ReadToEndAsync(),stderr=child.StandardError.ReadToEndAsync();
        try
        {
            await child.WaitForExitAsync().WaitAsync(TimeSpan.FromSeconds(6));Assert.Equal(admitted?0:4,child.ExitCode);Assert.Equal(admitted?"PUBLIC_ADMITTED":"PUBLIC_BLOCKED",(await stdout).Trim());Assert.Equal("",await stderr);output.WriteLine($"PUBLIC_ADMISSION_CHILD exit={child.ExitCode} driverSha256={Rig.Hash(Driver)} IO=Settled");
        }
        finally{if(!child.HasExited){child.Kill();await child.WaitForExitAsync().WaitAsync(TimeSpan.FromSeconds(3));}await Task.WhenAll(stdout,stderr);}
    }
    [Fact]public async Task UpgradeWaitsForActualOwnedAppAndBrokerExitThenStartsTheWholeSelectedGeneration()
    {
        using var r=new Rig();var previous=await r.Store.ApplyAsync(r.Package("1.0"),r.Pin("1.0"));r.Package("2.0");var data=r.DataBytes();var port=new Port(r,output);
        await using(var coordinator=r.Coordinator(port))
        {

            var upgraded=await coordinator.UpgradeAsync(Path.Combine(r.Root,"package-2.0"),r.Pin("2.0"));Assert.Equal(InstallationUpgradeOutcome.Upgraded,upgraded.Outcome);Assert.Null(upgraded.Failure);Assert.Equal("2.0",r.Store.ReadActive()!.Version);
            Assert.Equal(3,port.Children.Count);Assert.All(port.Children.Take(2),c=>{Assert.True(c.Settled);Assert.Equal(0,c.Exit);Assert.Contains("PUBLIC_INSTALL_EXIT",c.Lines);});Assert.False(port.Children[2].Process.HasExited);Assert.Equal(upgraded.Active!.DirectoryPath,port.Children[2].Directory);Assert.False(File.Exists(Path.Combine(r.App,"upgrade-pending.json")));
            Assert.Equal(data.OrderBy(p=>p.Key),r.DataBytes().OrderBy(p=>p.Key));await AdmissionChild(r,true);
        }
        port.AllSettled();
    }
