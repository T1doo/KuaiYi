    private sealed class UiQueue:SynchronizationContext
    {
        private readonly Queue<(SendOrPostCallback Callback,object? State)> work=[];
        public override void Post(SendOrPostCallback callback,object? state)=>work.Enqueue((callback,state));
        public void Drain(){while(work.TryDequeue(out var next))next.Callback(next.State);}
    }
