    private sealed class Destination:SynchronizationContext
    {
        private readonly ConcurrentQueue<(SendOrPostCallback,object?)> queue=new();public volatile bool Hold;
        public int Pending=>queue.Count;
        public override void Post(SendOrPostCallback action,object? state){if(Hold)queue.Enqueue((action,state));else action(state);}
        public void Drain(){while(queue.TryDequeue(out var item))item.Item1(item.Item2);}
    }
