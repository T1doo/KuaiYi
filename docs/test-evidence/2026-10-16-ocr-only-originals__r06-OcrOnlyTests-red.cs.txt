    private sealed class Immediate:SynchronizationContext{public override void Post(SendOrPostCallback action,object? state)=>action(state);}
    private sealed class Deferred:SynchronizationContext
    {
        private readonly ConcurrentQueue<Action> queue=new();public int Count=>queue.Count;
        public override void Post(SendOrPostCallback action,object? state)=>queue.Enqueue(()=>action(state));
        public void Drain(){while(queue.TryDequeue(out var callback))callback();}
    }
    private sealed class SyntheticOcr:P.ILocalOcrWorker
    {
        public P.ImageFailure Failure;public string Text="PUBLIC SYNTHETIC OCR TEXT";
        public bool Finalized=true,Disposed,Hold,WrongJob,InvalidBounds,FinalizeThrows;
        public Action? OnDispose=null;
        public TaskCompletionSource Entered=new(TaskCreationOptions.RunContinuationsAsynchronously),Release=new(TaskCreationOptions.RunContinuationsAsynchronously);
        public ReadOnlyMemory<byte> Borrowed;public CancellationToken Token;public int Calls;
        public async Task<P.ImageReply> RunAsync(ReadOnlyMemory<byte> image,P.ImageJob job,CancellationToken cancellation=default)
        {
            Calls++;Borrowed=image;Token=cancellation;Entered.TrySetResult();if(Hold)await Release.Task;
            return new(WrongJob?job with{Version=job.Version+1}:job,Failure,Failure==P.ImageFailure.None?new(8,8,job.Language,Text,InvalidBounds?[new("synthetic",[new("synthetic",8,0,double.Epsilon,1)])]:[]):null);
        }
        public ValueTask<bool> TryFinalizeAsync()=>FinalizeThrows?ValueTask.FromException<bool>(new IOException("synthetic cleanup failure")):ValueTask.FromResult(Finalized);
        public ValueTask DisposeAsync(){Disposed=true;OnDispose?.Invoke();return ValueTask.CompletedTask;}
    }

        public static async Task<Rig> Create(KuaiYi.Core.Settings.ApiProtocol protocol=KuaiYi.Core.Settings.ApiProtocol.OpenAIChat,SynchronizationContext? destination=null,bool enabled=true)
        {
            var r=new Rig{Protocol=protocol};await r.Server.InitializeAsync();r.Script=r.Server.Add("runs","application/json");
            var prefix=new Uri(r.Server.BaseUri,$"runs/{Guid.NewGuid():N}/fragmented/");var boundary=DevelopmentTranslationBoundary.FromPrefix(prefix);
            r.Settings=new(new(){ActiveProfileId="ocr-fixture",Profiles=[new(){Id="ocr-fixture",Protocol=protocol,BaseUrl=prefix.AbsoluteUri,Model=DevelopmentTranslationBoundary.FixtureModel,Authentication=ApiAuthentication.None,Streaming=false,Capabilities=new(){Vision=Capability.Unsupported}}]},"ocr-synthetic",SettingsLoadState.Current);
            r.Gate.ObserveSettingsMetadata(r.Settings);r.Gate.ConsentTo(new(prefix));if(enabled)Assert.True(r.Gate.TryEnable(new(prefix),true));r.Results=new(r.Gate);
            r.Local=new(r.Gate,()=>r.Settings,(version,source)=>{r.OnSourceCheck?.Invoke();return r.Current&&r.Version==version&&r.Image.SourceId==source;},destination??new Immediate(),()=>{r.Factories++;return r.Factory?.Invoke()??r.Worker;});
            r.Text=new(r.Gate,()=>r.Settings,(_,_)=>true,r.Provider,new Immediate(),r.Results,_=>{},developmentBoundary:boundary,ocrSourceCurrent:r.Local.IsApprovedSourceCurrent);
            return r;
        }

    private static void Invalidate(Rig rig,string change)
    {
        switch(change)
        {
            case "off":rig.Gate.Disable();break;case "locked":rig.Gate.SetSessionLocked(true);break;case "hidden":rig.Gate.Disable(hide:true);break;

    }
